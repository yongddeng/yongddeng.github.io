<script>
(function () {
  document.querySelectorAll('.nf-embed').forEach(function (el) {
    el.addEventListener('click', function () {
      var id = el.getAttribute('data-id');
      el.innerHTML = '<iframe src="https://www.youtube.com/embed/' + id +
        '?autoplay=1" title="video" allow="autoplay; encrypted-media; picture-in-picture" allowfullscreen></iframe>';
    });
  });

  var updates = [];
  document.querySelectorAll('.nf-row').forEach(function (row) {
    var track = row.querySelector('.nf-track');
    var ctrl = row.querySelector('.nf-ctrl');
    var prev = row.querySelector('.nf-prev');
    var next = row.querySelector('.nf-next');
    if (!track || !ctrl) return;
    function update() {
      var max = track.scrollWidth - track.clientWidth - 2;
      var show = max > 0 ? '' : 'none';
      prev.style.display = show;
      next.style.display = show;
      prev.disabled = track.scrollLeft <= 0;
      next.disabled = track.scrollLeft >= max;
    }
    prev.addEventListener('click', function () { track.scrollBy({ left: -track.clientWidth * 0.8, behavior: 'smooth' }); });
    next.addEventListener('click', function () { track.scrollBy({ left: track.clientWidth * 0.8, behavior: 'smooth' }); });
    track.addEventListener('scroll', update, { passive: true });
    updates.push(update);
    update();
  });

  // One window listener, replaced on SPA revisits — per-row listeners
  // would accumulate because this script re-runs on every navigation.
  if (window.nfOnResize) window.removeEventListener('resize', window.nfOnResize);
  window.nfOnResize = function () { updates.forEach(function (u) { u(); }); };
  window.addEventListener('resize', window.nfOnResize);
})();
</script>
