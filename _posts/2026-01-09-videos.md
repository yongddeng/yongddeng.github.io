<script>
(function () {
  var screen = document.getElementById('vp-screen');
  var titleEl = document.getElementById('vp-title');
  var durEl = document.getElementById('vp-dur');
  var list = document.getElementById('vp-list');
  var items = Array.prototype.slice.call(list.querySelectorAll('.vp-item'));
  var current = null;
  var playing = false;

  function select(item, autoplay) {
    current = item;
    playing = autoplay;
    items.forEach(function (i) { i.classList.toggle('on', i === item); });
    titleEl.textContent = item.getAttribute('data-title');
    durEl.textContent = item.getAttribute('data-dur');
    var id = item.getAttribute('data-id');
    if (autoplay) {
      screen.innerHTML = '<iframe src="https://www.youtube.com/embed/' + id +
        '?autoplay=1" title="video" allow="autoplay; encrypted-media; picture-in-picture" allowfullscreen></iframe>';
    } else {
      screen.innerHTML = '<img class="vp-thumb" src="https://i.ytimg.com/vi/' + id +
        '/hqdefault.jpg" alt="" loading="lazy"><span class="vp-play"></span>';
    }
  }

  list.addEventListener('click', function (e) {
    var item = e.target.closest('.vp-item');
    if (item) select(item, false);
  });
  screen.addEventListener('click', function () {
    if (!playing && current) select(current, true);
  });

  function step(delta) {
    var i = (items.indexOf(current) + delta + items.length) % items.length;
    select(items[i], true);
  }
  document.getElementById('vp-prev').addEventListener('click', function () { step(-1); });
  document.getElementById('vp-next').addEventListener('click', function () { step(1); });
  document.getElementById('vp-playbtn').addEventListener('click', function () {
    if (current) select(current, true);
  });
  document.getElementById('vp-stop').addEventListener('click', function () {
    if (current) select(current, false);
  });

  select(items[0], false);
})();
</script>
